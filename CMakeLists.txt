cmake_minimum_required(VERSION 3.24)
project(greenctx_benchmark_v3 LANGUAGES CXX CUDA)
if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.1)
 message(FATAL_ERROR "Green Context Runtime API requires CUDA Toolkit 13.1+")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(CUDAToolkit 13.1 REQUIRED)
add_executable(greenctx_bench src/main.cu)
target_include_directories(greenctx_bench PRIVATE include)
target_link_libraries(greenctx_bench PRIVATE CUDA::cudart)
set_target_properties(greenctx_bench PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
