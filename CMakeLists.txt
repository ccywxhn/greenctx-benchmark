cmake_minimum_required(VERSION 3.24)
project(greenctx_benchmark_v3 LANGUAGES CXX CUDA)
if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.1)
 message(FATAL_ERROR "CUDA Toolkit 13.1+ is required for this Driver API benchmark")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
find_package(CUDAToolkit 13.1 REQUIRED)
set(KERNEL_PTX "${CMAKE_CURRENT_BINARY_DIR}/greenctx_kernels.ptx")
add_custom_command(OUTPUT "${KERNEL_PTX}" COMMAND "${CMAKE_CUDA_COMPILER}" --ptx --std=c++17 --gpu-architecture=compute_75 "${CMAKE_CURRENT_SOURCE_DIR}/src/kernels.cu" -o "${KERNEL_PTX}" DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/kernels.cu" VERBATIM)
add_custom_target(greenctx_ptx DEPENDS "${KERNEL_PTX}")
add_executable(greenctx_bench src/main.cpp)
add_dependencies(greenctx_bench greenctx_ptx)
target_include_directories(greenctx_bench PRIVATE include)
target_compile_definitions(greenctx_bench PRIVATE GCB_KERNEL_PTX_PATH=\"${KERNEL_PTX}\")
target_link_libraries(greenctx_bench PRIVATE CUDA::cuda_driver)
